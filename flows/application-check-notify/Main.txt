# 手順1. Edge 起動・ログイン
BLOCK Step1_Login
ON BLOCK ERROR
    GOTO ErrorEnd
END
    SET StepMessage TO $'''手順1-1〜1-2：モック画面に接続できません。server.js が起動しているか確認してください'''
    WebAutomation.LaunchEdge.LaunchEdge Url: $'''%BaseUrl%/login.html''' WindowState: WebAutomation.BrowserWindowState.Maximized ClearCache: False ClearCookies: False WaitForPageToLoadTimeout: 60 Timeout: 60 BrowserInstance=> Browser
    WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['LoginPage']['login-id']) FOR 30
    SET StepMessage TO $'''手順1-3〜1-4：ログイン画面で ID・パスワードの入力またはログインボタンのクリックができませんでした'''
    # Edge 起動直後のログインの失敗の対策（04c・04d）：ボタンまで表示されるのを待ってから、次を最大3回くり返す（一覧画面が表示されたら抜ける）
    # Edge のウィンドウを前面に出す → ID・パスワードの入力 → 入力欄の値を確認 → 確認できたらログインボタンをクリック → 一覧画面の読み込み完了を待つ（30 秒）→ 一覧画面が表示されたかを確認
    # 前面に出す・入力・確認・クリック・一覧の待機の1回ごとのエラーは、その場では止めずに次の回でやり直す。3回とも一覧画面が表示されなければ E2（入力できない）または E3（一覧が出ない）で止まる。ログインボタン・「詳細」・「一覧へ戻る」のクリックは「物理クリックを送信」をオフ（MouseClick: False。マウス カーソルを動かさない）にしている（04e。verification-log.md 10-0j）
    WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['LoginPage']['login-button']) FOR 30
    SET LoginCheck TO $'''NG'''
    SET ListCheck TO $'''NG'''
    LOOP LoginTry FROM 1 TO 3 STEP 1
        IF ListCheck <> $'''OK''' THEN
            IF LoginTry > 1 THEN
                WAIT 2
            END
            # フローが起動した Edge のウィンドウ（ブラウザー インスタンス Browser の Handle プロパティ＝ウィンドウ ハンドル）を前面に出す。失敗しても止めない
            UIAutomation.FocusWindow.FocusByInstanceOrHandle WindowInstance: Browser.Handle
            ON ERROR
            END
            SET LoginCheck TO $'''NG'''
            WebAutomation.PopulateTextField.PopulateTextField BrowserInstance: Browser Control: appmask['LoginPage']['login-id'] Text: LoginId EmulateTyping: True UnfocusAfterPopulate: False Mode: WebAutomation.PopulateTextMode.Replace WaitForPageToLoadTimeout: 60
            ON ERROR
            END
            WebAutomation.PopulateTextField.PopulateTextField BrowserInstance: Browser Control: appmask['LoginPage']['login-password'] Text: LoginPassword EmulateTyping: True UnfocusAfterPopulate: False Mode: WebAutomation.PopulateTextMode.Replace WaitForPageToLoadTimeout: 60
            ON ERROR
            END
            WebAutomation.ExecuteJavascript BrowserInstance: Browser Javascript: $'''function ExecuteScript() { var id = document.getElementById(\"login-id\"); var pw = document.getElementById(\"login-password\"); if (id && pw && id.value === \"%LoginId%\" && pw.value === \"%LoginPassword%\") { return \"OK\"; } return \"NG\"; }''' Result=> LoginCheck
            ON ERROR
            END
            IF LoginCheck = $'''OK''' THEN
                WebAutomation.Click.Click BrowserInstance: Browser Control: appmask['LoginPage']['login-button'] ClickType: WebAutomation.ClickType.LeftClick MouseClick: False WaitForPageToLoadTimeout: 60 MousePositionRelativeToElement: WebAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
                ON ERROR
                END
                WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['ListPage']['application-table-loaded']) FOR 30
                ON ERROR
                END
            END
            # 一覧画面の読み込みが完了していれば（前の回のクリックで遅れて表示された場合も含む）やり直しを抜ける
            IF (WebAutomation.IfWebPageContains.WebPageContainsElement BrowserInstance: Browser Control: appmask['ListPage']['application-table-loaded']) THEN
                SET ListCheck TO $'''OK'''
            END
        END
    END
    IF ListCheck <> $'''OK''' THEN
        IF LoginCheck <> $'''OK''' THEN
            SET StepMessage TO $'''手順1-3〜1-4：ログイン画面で ID・パスワードを入力できませんでした（3 回試しても一覧画面に進めず、最後の回は入力欄に値が入りませんでした）'''
            GOTO ErrorEnd
        END
        SET StepMessage TO $'''手順1-4〜1-5：ログインボタンをクリックしても一覧画面が表示されませんでした（3 回試しました。クリックに失敗したか、30 秒待っても一覧の読み込みが完了しません）。ID・パスワードと一覧画面の表示を確認してください'''
        GOTO ErrorEnd
    END
    # ここに来るのは一覧画面が表示された場合だけ。念のため一覧の読み込み完了をもう一度確かめてから手順2 へ進む（表示済みなのですぐに終わる）
    SET StepMessage TO $'''手順1-5：ログイン後に一覧画面が表示されませんでした（30 秒待っても読み込みが完了しません）。ID・パスワードと一覧画面の表示を確認してください'''
    WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['ListPage']['application-table-loaded']) FOR 30
END
# 手順2. 一覧から申込番号を全件取得
BLOCK Step2_GetList
ON BLOCK ERROR
    GOTO ErrorEnd
END
    SET StepMessage TO $'''手順2-1：一覧から申込番号を取得できませんでした'''
    WebAutomation.ExecuteJavascript BrowserInstance: Browser Javascript: $'''function ExecuteScript() { var rows = document.querySelectorAll(\"#application-tbody tr[data-application-no]\"); var nos = []; for (var i = 0; i < rows.length; i++) { nos.push(rows[i].getAttribute(\"data-application-no\")); } return nos.join(\",\"); }''' Result=> ApplicationNosText
    Text.SplitText.SplitWithDelimiter Text: ApplicationNosText CustomDelimiter: $''',''' IsRegEx: False Result=> ApplicationNos
    SET StepMessage TO $'''手順2-2：一覧の件数表示（「全 N 件」）を取得できませんでした'''
    # 件数表示の文字（例「全 20 件」）を Own Text で読み、数字の部分だけ取り出して数値にする（data-count 属性は PAD では空の文字が返るため）
    WebAutomation.GetDetailsOfElement BrowserInstance: Browser Control: appmask['ListPage']['application-count'] AttributeName: $'''Own Text''' AttributeValue=> DataCountText
    Text.ParseText.RegexParseForFirstOccurrence Text: DataCountText TextToFind: $'''[0-9]+''' StartingPosition: 0 IgnoreCase: False OccurrencePosition=> DataCountPosition Match=> DataCountDigits
    Text.ToNumber Text: DataCountDigits Number=> DataCount
    IF ApplicationNos.Count <> DataCount THEN
        SET StepMessage TO $'''手順2-2：件数が一致しません（画面の件数 %DataCount% 件／取得 %ApplicationNos.Count% 件）。取得した申込番号：%ApplicationNosText%'''
        GOTO ErrorEnd
    END
END
# 手順3. 各案件の詳細を取得（手順2 で取得した申込番号リストの全件を繰り返す）
BLOCK Step3_GetDetails
ON BLOCK ERROR
    GOTO ErrorEnd
END
    LOOP FOREACH CurrentNo IN ApplicationNos
        SET StepMessage TO $'''手順3-1：申込番号 %CurrentNo% の「詳細」ボタンをクリックできませんでした'''
        UIAutomation.FocusWindow.FocusByInstanceOrHandle WindowInstance: Browser.Handle
        ON ERROR
        END
        WebAutomation.Click.Click BrowserInstance: Browser Control: appmask['ListPage']['detail-button'] ClickType: WebAutomation.ClickType.LeftClick MouseClick: False WaitForPageToLoadTimeout: 60 MousePositionRelativeToElement: WebAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
        SET StepMessage TO $'''手順3-2：申込番号 %CurrentNo% の詳細画面が表示されませんでした（30 秒。該当する申込が見つからない場合を含む）'''
        WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['DetailPage']['application-detail-loaded']) FOR 30
        SET StepMessage TO $'''手順3-3〜3-4：申込番号 %CurrentNo% の詳細画面から項目を取得できませんでした'''
        # 14項目（#field-<キー>）の表示テキストを JavaScript 1回でまとめて取得する（要素ごとに14回取得すると1件あたり約26〜30秒かかったため）。値はカンマでつなぐ（値にカンマは含まれない：docs/screens.md 1 のデータの制約）
        WebAutomation.ExecuteJavascript BrowserInstance: Browser Javascript: $'''function ExecuteScript() { var keys = [\"applicationNo\", \"applicationDate\", \"customerName\", \"supplyPointNo\", \"applicationType\", \"capacityBefore\", \"capacityAfter\", \"status\", \"assignee\", \"siteAddress\", \"desiredDate\", \"scheduledWorkDate\", \"receptionNo\", \"remarks\"]; var vals = []; for (var i = 0; i < keys.length; i++) { var el = document.getElementById(\"field-\" + keys[i]); if (!el) { return \"NG:field-\" + keys[i]; } vals.push(el.textContent.trim()); } return vals.join(\",\"); }''' Result=> DetailText
        Text.SplitText.SplitWithDelimiter Text: DetailText CustomDelimiter: $''',''' IsRegEx: False Result=> DetailValues
        IF DetailValues.Count <> 14 THEN
            SET StepMessage TO $'''手順3-4：申込番号 %CurrentNo% の詳細画面から14項目を取得できませんでした（取得できた値は %DetailValues.Count% 個）'''
            GOTO ErrorEnd
        END
        IF DetailValues[0] <> CurrentNo THEN
            SET StepMessage TO $'''手順3-3：申込番号が一致しません（対象 %CurrentNo%／詳細画面 %DetailValues[0]%）'''
            GOTO ErrorEnd
        END
        # 14項目を docs/screens.md 1-2 の順に並べたリストのまま1行追加（空の備考も空の値として位置で入るので列がずれない）
        Variables.AddRowToDataTable.AppendRowToDataTable DataTable: CurrentTable RowToAdd: DetailValues
        SET StepMessage TO $'''手順3-5：申込番号 %CurrentNo% の詳細画面から一覧画面に戻れませんでした（30 秒）'''
        UIAutomation.FocusWindow.FocusByInstanceOrHandle WindowInstance: Browser.Handle
        ON ERROR
        END
        WebAutomation.Click.Click BrowserInstance: Browser Control: appmask['DetailPage']['back-to-list-button'] ClickType: WebAutomation.ClickType.LeftClick MouseClick: False WaitForPageToLoadTimeout: 60 MousePositionRelativeToElement: WebAutomation.RectangleEdgePoint.MiddleCenter OffsetX: 0 OffsetY: 0
        WAIT (WebAutomation.WaitForWebPageContent.WebPageToContainElement BrowserInstance: Browser Control: appmask['ListPage']['application-table-loaded']) FOR 30
    END

# 手順5. 前回との比較と SharePoint への保存（資産5で追加）
# 手順5-1. SharePoint から前回の latest.csv を取得する（今回の latest.csv を保存する 5-3 より前に行う）
BLOCK Step5a_GetPrevious
ON BLOCK ERROR
    GOTO ErrorEnd
END
    SET StepMessage TO $'''手順5-1：SharePoint から前回の latest.csv を取得できませんでした（ファイルが無いのとは別の理由です。初回扱いにはしません）。SharePoint の接続・保存先・ネットワークを確認してください'''
    SET PreviousExists TO False
    SET PreviousNosKey TO $''','''
    # 取得に成功すると PreviousBinary に latest.csv の内容（バイナリ）が入る。失敗した場合は NONE のまま
    SET PreviousBinary TO $'''NONE'''
    @@dataset: '<サイト名> - <保存先サイト>'
    @@path: '%SharePointFolder%/latest.csv'
    @@connectionDisplayName: '<SharePoint 接続参照名>'
    External.InvokeCloudConnector Connection: '<SharePoint 接続参照ID>' ConnectorId: '/providers/Microsoft.PowerApps/apis/shared_sharepointonline' OperationId: 'GetFileContentByPath' @dataset: $'''<保存先サイト>''' @path: $'''%SharePointFolder%/latest.csv''' @inferContentType: True @GetFileContentByPathResponse=> PreviousBinary
    ON ERROR
    END
    # 失敗の内容を残す（Reset: False。「ファイルが無い」以外の失敗なら、ErrorEnd でも同じエラー情報を表示する）
    ERROR => PrevError Reset: False
    IF PreviousBinary = $'''NONE''' THEN
        # 取得できなかった。ファイルが無い（初回）のか、別の理由（接続エラー等）かを、エラー情報の文面で見分ける
        SET PrevErrorText TO $'''%PrevError%'''
        IF Contains(PrevErrorText, $'''404''', True) OR Contains(PrevErrorText, $'''NotFound''', True) OR Contains(PrevErrorText, $'''not found''', True) OR Contains(PrevErrorText, $'''見つかりません''', True) OR Contains(PrevErrorText, $'''存在しません''', True) THEN
            # ファイルが無い＝初回。エラーにせず、比較しない（PreviousExists は False のまま）
            SET PreviousExists TO False
        ELSE
            # 「無い」以外の失敗は初回扱いにしない（latest.csv を誤って上書きしないため）。手順5-1 で停止する
            GOTO ErrorEnd
        END
    ELSE
        SET StepMessage TO $'''手順5-1：前回の latest.csv は取得できましたが、内容を読み込めませんでした'''
        File.ConvertFromBinary BinaryData: PreviousBinary File: $'''%WorkFolder%\\previous.csv''' IfFileExists: File.IfExists.Overwrite
        File.ReadFromCSVFile.ReadCSV CSVFile: $'''%WorkFolder%\\previous.csv''' Encoding: File.CSVEncoding.UTF8 TrimFields: False FirstLineContainsColumnNames: True ColumnsSeparator: File.CSVColumnsSeparator.Comma CSVTable=> PreviousTable
        Variables.RetrieveDataTableColumnIntoList DataTable: PreviousTable ColumnNameOrIndex: $'''申込番号''' ColumnAsList=> PreviousNos
        # 比べやすいように ,申込番号,申込番号,…, の形の文字にする（前後と間をカンマで囲む）
        LOOP FOREACH PreviousNo IN PreviousNos
            SET PreviousNosKey TO PreviousNosKey + PreviousNo + $''','''
        END
        SET PreviousExists TO True
    END
END

# 手順5-3. SharePoint への保存（履歴ファイル → latest.csv の順。ファイルの内容はバイナリで渡す）
BLOCK Step5c_Save
ON BLOCK ERROR
    GOTO ErrorEnd
END
    SET StepMessage TO $'''手順5-3：SharePoint への履歴ファイルの保存に失敗しました（latest.csv は保存していません）'''
    File.ConvertToBinary File: $'''%WorkFolder%\\%HistoryFileName%''' BinaryData=> HistoryBinary
    @@dataset: '<サイト名> - <保存先サイト>'
    @@folderPath: '%SharePointFolder%'
    @@connectionDisplayName: '<SharePoint 接続参照名>'
    External.InvokeCloudConnector Connection: '<SharePoint 接続参照ID>' ConnectorId: '/providers/Microsoft.PowerApps/apis/shared_sharepointonline' OperationId: 'CreateFile' @dataset: $'''<保存先サイト>''' @folderPath: SharePointFolder @name: HistoryFileName @body: HistoryBinary @CreateFileResponse=> CreateFileResponse2
    # 履歴ファイルへのリンク（サイトの URL ＋ フォルダー ＋ ファイル名。空白は %20 にする）
    SET HistoryFileLink TO $'''%SharePointSiteUrl%/%SharePointFolder%/%HistoryFileName%'''
    Text.Replace.ReplaceTextWithRegex Text: HistoryFileLink TextToFind: $'''\\x20''' IgnoreCase: False ReplaceWith: $'''%%20''' ActivateEscapeSequences: False Result=> HistoryFileLink
    SET StepMessage TO $'''手順5-3：履歴ファイルは SharePoint に保存しましたが、latest.csv の保存に失敗しました（latest.csv は前回のまま、または無い状態です）'''
    File.ConvertToBinary File: $'''%WorkFolder%\\latest.csv''' BinaryData=> LatestBinary
    @@dataset: '<サイト名> - <保存先サイト>'
    @@folderPath: '%SharePointFolder%'
    @@connectionDisplayName: '<SharePoint 接続参照名>'
    External.InvokeCloudConnector Connection: '<SharePoint 接続参照ID>' ConnectorId: '/providers/Microsoft.PowerApps/apis/shared_sharepointonline' OperationId: 'CreateFile' @dataset: $'''<保存先サイト>''' @folderPath: SharePointFolder @name: $'''latest.csv''' @body: LatestBinary @CreateFileResponse=> CreateFileResponse
END
# 手順6. Teams 投稿（毎回必ず1回投稿する。初回／新規あり／新規0件で文言を変える。投稿先は作成者本人の個人チャット）
BLOCK Step6_Notify
ON BLOCK ERROR
    GOTO ErrorEnd
END
    # ここから先で失敗しても、失敗通知は投稿しない（画面のメッセージだけ出して終了。CSV は SharePoint に保存済み）
    SET NotifyOnError TO $'''NO'''
    SET StepMessage TO $'''手順6：Teams への投稿に失敗しました。CSV は SharePoint に保存済みです（このまま再実行すると、新規案件は 0 件になります）'''
    Text.ConvertDateTimeToText.FromCustomDateTime DateTime: RunDateTime CustomFormat: $'''yyyy/MM/dd HH:mm''' Result=> RunDateTimeText
    IF PreviousExists = False THEN
        SET PostPattern TO $'''初回'''
        SET PostMessage TO $'''【申込状況チェック】初回実行のため、全 %CurrentTable.RowsCount% 件を保存しました（比較する前回データがありません）（取得日時：%RunDateTimeText%）<br><br>CSV：%HistoryFileLink%'''
    ELSE
        IF NewRows.RowsCount > 0 THEN
            SET PostPattern TO $'''新規あり'''
            # 列の番号：0＝申込番号、2＝お客さま名、4＝申込種別、5＝変更前容量、6＝変更後容量、7＝ステータス（docs/screens.md 1-2 の順）
            SET NewLines TO $''''''
            SET NewIndex TO 0
            LOOP FOREACH NewRow IN NewRows
                SET NewIndex TO NewIndex + 1
                SET NewLine TO $'''%NewIndex%. %NewRow[0]%／%NewRow[2]%／%NewRow[4]%／%NewRow[5]% → %NewRow[6]%／%NewRow[7]%<br>'''
                SET NewLines TO NewLines + NewLine
            END
            SET PostMessage TO $'''【申込状況チェック】新規案件が %NewRows.RowsCount% 件あります（取得日時：%RunDateTimeText%／全 %CurrentTable.RowsCount% 件）<br><br>%NewLines%<br>CSV：%HistoryFileLink%'''
